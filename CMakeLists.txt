# Two packages come out of this one tree, and they are built separately.
#
# The application is sandboxed and Store-eligible. The notification daemon
# cannot be: observing other applications' notifications needs a D-Bus
# monitor connection, which the Sailjail proxy will not relay. Rather than
# give that up for the whole app, the daemon is its own package, built with
# -DBUILD_APP=OFF -DBUILD_DAEMON=ON from rpm/suuntosync-notifyd.spec.
#
# Separate options rather than one switch, so that a developer build can
# have both at once.
option(BUILD_APP "Build the Silica application" ON)
option(BUILD_DAEMON "Build the notification daemon" OFF)

# NOTE: no "Bluetooth" component - Sailfish OS 5.1's SDK target has no
# pkgconfig(Qt5Bluetooth) at all (confirmed by a real build attempt: zypper
# found no provider, by name or capability). QtConnectivity/QtBluetooth isn't
# part of the Sailfish OS Qt5 distribution; BLE work here goes through BlueZ
# 5's own D-Bus API (org.bluez.*) via Qt5::DBus instead (added below, Phase 5) -
# see ~/.claude/plans/agile-hopping-harp.md.
set(QT_COMPONENTS Core Sql Network DBus)
if(BUILD_APP)
    list(APPEND QT_COMPONENTS Qml Gui Quick Concurrent LinguistTools)
endif()
if(BUILD_DAEMON)
    # Contacts, for the name behind a ringing call's number. Only the daemon
    # needs it, and only because the daemon is the half that sees calls - see
    # src/notify/contactlookup.h for the privilege that makes it readable.
    list(APPEND QT_COMPONENTS Contacts)
endif()
find_package(Qt5 COMPONENTS ${QT_COMPONENTS} REQUIRED)

include(FindPkgConfig)
if(BUILD_APP)
    pkg_search_module(SAILFISH sailfishapp REQUIRED)
    pkg_search_module(SAILFISHSECRETS sailfishsecrets REQUIRED)
endif()

    # addressed to us, and a notification is addressed to the notification
    # server. See src/notify/notificationmonitor.h.
    pkg_search_module(DBUS1 dbus-1 REQUIRED)
endif()

# Sources are added back in as their own phases land (see
# ~/.claude/plans/agile-hopping-harp.md) so each build stays small and
# attributable to one phase if it breaks. Phase 2a: TokenVault (Sailfish
# Secrets) + CloudAccountStore (SQLite metadata), no network yet. Phase 5:
# BlueZ D-Bus device discovery/pairing. Phase 6: mdswirecodec (already
# separately unit-tested via tests/test_mdswirecodec.cpp, plain g++) +
# mdswhiteboardclient wiring it up to real GATT characteristics. Phase 6
# continued: sbemcontainer (already separately unit-tested via
# tests/test_sbemcontainer.cpp, plain g++) decodes the actual
# /Logbook/byId/<id>/Data payload once reassembled - Heatshrink
# decompression (vendored, ISC-licensed, src/ble/heatshrink/) followed by
# an SBEM0103 TLV walk. See docs/logbook-data-format.md. Phase 6 continued
# again: logbookdecoder (tests/test_logbookdecoder.cpp, plain g++) turns
# the decoded chunks into workout summary fields - GPS/HR/activity/cadence
# confirmed against Jarno's own FIT exports, plus altitude and derived
# ascent/descent (found once the APK yielded the watch's own field map -
# see the doc). Phase 6 continued
# once more: MdsWhiteboardClient::fetchLogbookData() and
# AppController::testLogbookFetch() wire logbookdecoder up to a real BLE
# fetch - marked experimental throughout (see fetchLogbookData()'s and
# Mds::encodeStreamStartTrigger()'s doc comments) since it replaces the
# real captured trigger sequence's elaborate handle-walk with a much
# simpler hypothesis, since confirmed on real hardware, and wired into
# WorkoutStore. Phase 6
# continued yet again: logentriesdecoder (tests/test_logentriesdecoder.cpp,
# plain g++) + Mds::encodeEntriesFetchTrigger() +
# MdsWhiteboardClient::fetchLogEntries() - a *different* shortcut for
# /Entries specifically (a TYPE=0x0D handle-fetch, not a TYPE=0x10 stream
# trigger), re-derived from decompiling libmds.so's own protocol_v9
# structure-deserializer code after the /Data-style shortcut was confirmed
# not to work for /Entries on real hardware. See
# docs/logbook-data-format.md's "Third decompilation pass" section;
# confirmed working on real hardware. And finally summarydecoder
# (tests/test_summarydecoder.cpp) + Mds::encodePagedReadRequest() +
# MdsWhiteboardClient::fetchSummary(): /Summary is a third transport again,
# fixed-size pages rather than a stream, carrying the watch's own exact
# totals (ascent, descent, energy, moving time) that /Data can only
# approximate. Finally sbemdescriptors (generated by
# tools/generate_sbem_tables.py from the watch's own captured descriptor
# table) + smldecoder: a generic, descriptor-driven decoder that reads any
# field in a payload by looking it up rather than by hard-coded offset -
# the way the official app does it. The two hand-written decoders above stay
# as they are, being hardware-validated; this covers everything they don't.
# Everything the daemon also needs. Named once so the two targets cannot
# drift apart: a decoder fixed for the app is fixed for the daemon.

# -DDAEMON_RPM=<path to suuntosync-notifyd-*.rpm> embeds that package in the
# application binary as a Qt resource, and the application grows a Settings
# entry that writes it out and asks the system to install it.
#
# Why that is worth doing: the daemon cannot be sandboxed, so it cannot be
# a Store package, and an application in the Store therefore cannot offer
# notifications at all. Shipping the daemon's RPM *inside* the application
# and handing it to the OS's own installer is how harbour-situations2
# application does it - verified by reading the installed binary, which
# carries a 32 kB binary RPM at a Qt resource path and links exactly one
# relevant symbol, QDesktopServices::openUrl. The application installs
# nothing itself; it opens a file and the system asks the user.
#
# Optional on purpose. The daemon's RPM is the product of a *separate* build
# of this same tree (-DBUILD_APP=OFF -DBUILD_DAEMON=ON, its own spec), so
# requiring it here would make the application unbuildable on its own and
# the ordering circular. Without it the application builds exactly as
# before and the Settings entry is absent rather than broken.
set(DAEMON_RPM "" CACHE FILEPATH
    "suuntosync-notifyd RPM to embed in the application for in-app install")

# A conventional path as well as the option, because the package build does
# not reach the cmake command line: rpm/harbour-suuntosync.spec calls
# %cmake, and threading an option through it would mean an rpm macro, an
# sfdk argument and a CI incantation that all have to agree. Dropping the
# file here instead needs none of that - the spec is untouched and the
# tarball carries it.
#
# CI builds the daemon's package first and copies it to this path; see
# .github/workflows/build.yml. Gitignored: a built RPM has no business in
# the source tree, it is only passing through.
if(NOT DAEMON_RPM AND EXISTS "${CMAKE_SOURCE_DIR}/daemon/embed/suuntosync-notifyd.rpm")
    set(DAEMON_RPM "${CMAKE_SOURCE_DIR}/daemon/embed/suuntosync-notifyd.rpm")
endif()

if(DAEMON_RPM)
    if(NOT EXISTS "${DAEMON_RPM}")
        message(FATAL_ERROR "DAEMON_RPM does not exist: ${DAEMON_RPM}")
    endif()
    # Read the first four bytes and insist on the RPM lead's magic
    # (ed ab ee db). A path typo that pointed at a tarball would otherwise
    # produce an application that offers an install and hands the installer
    # something it cannot read - a failure that would only show up on a
    # phone, which is the expensive place to find it.
    file(READ "${DAEMON_RPM}" DAEMON_RPM_LEAD HEX LIMIT 4)
    if(NOT DAEMON_RPM_LEAD STREQUAL "edabeedb")
        message(FATAL_ERROR
                "DAEMON_RPM is not an RPM (lead is 0x${DAEMON_RPM_LEAD}, "
                "expected 0xedabeedb): ${DAEMON_RPM}")
    endif()

    # The .qrc is generated rather than committed, because it has to name a
    # path only known at configure time, and the RPM is copied beside it
    # under a stable name so the resource path does not carry a version.
    #
    # "daemon-embed", not "rpms": sfdk builds in the source directory, so
    # this lands next to rpm/ and a one-letter difference between a source
    # directory and a generated one is a trap. The *resource* path is set by
    # the prefix below and does not follow this name - it stays /rpms, which
    # is what AppController reads.
    configure_file("${DAEMON_RPM}"
                    "${CMAKE_CURRENT_BINARY_DIR}/daemon-embed/suuntosync-notifyd.rpm"
                    COPYONLY)
    # The <file> path is relative to the .qrc, so both land in the same
    # generated directory.
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/daemon-embed/daemon-rpm.qrc"
            "<RCC>\n"
            "  <qresource prefix=\"/rpms\">\n"
            "    <file>suuntosync-notifyd.rpm</file>\n"
            "  </qresource>\n"
            "</RCC>\n")

    qt5_add_resources(DAEMON_RPM_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/daemon-embed/daemon-rpm.qrc")
    target_sources(harbour-suuntosync PRIVATE ${DAEMON_RPM_RESOURCE})
    target_compile_definitions(harbour-suuntosync PRIVATE SUUNTOSYNC_EMBEDDED_DAEMON_RPM)
    message(STATUS "Embedding the notification daemon: ${DAEMON_RPM}")
endif()

# Source strings (qsTr()/tr()/QCoreApplication::translate()) are English;
# translations/harbour-suuntosync.ts tracks them as the extraction base (kept
# up to date by manually running lupdate, not as part of this build) and each
# translations/harbour-suuntosync-<lang>.ts is a translated copy of it.
# qt5_add_translation only compiles .ts -> .qm (lrelease) - it does not run
# lupdate - so a normal build never touches the committed .ts files.
file(GLOB TS_FILES "${CMAKE_SOURCE_DIR}/translations/*.ts")
file(GLOB LANG_TS_FILES "${CMAKE_SOURCE_DIR}/translations/harbour-suuntosync-*.ts")
qt5_add_translation(QM_FILES ${LANG_TS_FILES})
add_custom_target(translations ALL DEPENDS ${QM_FILES})
add_dependencies(harbour-suuntosync translations)

install(FILES ${QM_FILES}
    DESTINATION share/harbour-suuntosync/translations
)
